cmake_minimum_required(VERSION 3.25)
project(ravet_vst3_host LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "This native module loader has local macOS evidence only")
endif()

if(NOT EXISTS "${RAVET_VST3_SDK}/CMakeLists.txt")
    message(FATAL_ERROR "Set RAVET_VST3_SDK to a caller-owned external VST3 SDK")

set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${RAVET_VST3_SDK}" sdk-build)
add_library(ravet-vst3-host STATIC provider.cpp
    "${RAVET_VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm")
target_compile_features(ravet-vst3-host PUBLIC cxx_std_17)
target_compile_options(ravet-vst3-host PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_include_directories(ravet-vst3-host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(ravet-vst3-host SYSTEM PRIVATE "${RAVET_VST3_SDK}")
target_link_libraries(ravet-vst3-host PUBLIC sdk_hosting "-framework Cocoa" "-framework CoreFoundation")
set_source_files_properties(provider.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
configure_file("${RAVET_VST3_SDK}/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/vst3-sdk-LICENSE.txt" COPYONLY)

        foreach(profile IN ITEMS normal fast)
            set(target "ravet-vst3-${name}-${profile}-tests")

            target_include_directories(${target} SYSTEM PRIVATE "${RAVET_VST3_SDK}")

                target_compile_options(${target} PRIVATE -ffast-math)
            endif()

            add_test(NAME ${target} COMMAND ${target})
        endforeach()

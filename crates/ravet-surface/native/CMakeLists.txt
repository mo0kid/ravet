if(APPLE)
    enable_language(OBJCXX)
    add_library(ravet-web-editor STATIC RavetWebView.mm RavetFileDialog.mm)
    target_compile_options(ravet-web-editor PRIVATE -fobjc-arc)
    # Distinct Objective-C class names per build tree; see RavetObjcNames.h.
    string(SHA1 RAVET_OBJC_HASH "${CMAKE_BINARY_DIR}")
    string(SUBSTRING "${RAVET_OBJC_HASH}" 0 12 RAVET_OBJC_HASH)
    string(MAKE_C_IDENTIFIER "${CMAKE_PROJECT_NAME}_${RAVET_OBJC_HASH}" RAVET_OBJC_DEFAULT)
    set(RAVET_OBJC_SUFFIX "${RAVET_OBJC_DEFAULT}" CACHE STRING "Suffix for this binary's Objective-C class names")
    target_compile_definitions(ravet-web-editor PRIVATE RAVET_OBJC_SUFFIX=${RAVET_OBJC_SUFFIX})
    target_link_libraries(ravet-web-editor PUBLIC "-framework Cocoa" "-framework WebKit" "-framework UniformTypeIdentifiers")
elseif(WIN32)
    set(RAVET_WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet SDK")
    if(NOT EXISTS "${RAVET_WEBVIEW2_SDK}/build/native/include/WebView2.h")
        message(FATAL_ERROR "Set RAVET_WEBVIEW2_SDK to the extracted WebView2 SDK")
    endif()
    add_library(ravet-web-editor STATIC RavetWebViewWindows.cpp RavetFileDialogWindows.cpp)
    target_include_directories(ravet-web-editor PRIVATE "${RAVET_WEBVIEW2_SDK}/build/native/include")
    if(MINGW)
        target_include_directories(ravet-web-editor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mingw")
    endif()
    target_compile_definitions(ravet-web-editor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(ravet-web-editor PUBLIC ole32 user32 shell32 shlwapi uuid comdlg32)
elseif(UNIX)
    option(RAVET_WEB_LINUX_SOFTWARE_RENDERING "Use non-accelerated WebKit painting for Linux editors" OFF)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1>=2.40)
    pkg_check_modules(JSON_GLIB REQUIRED IMPORTED_TARGET json-glib-1.0)
    add_library(ravet-web-editor STATIC RavetWebViewLinux.cpp RavetFileDialogLinux.cpp)
    target_compile_definitions(ravet-web-editor PRIVATE RAVET_WEB_LINUX_SOFTWARE_RENDERING=$<BOOL:${RAVET_WEB_LINUX_SOFTWARE_RENDERING}>)
    target_link_libraries(ravet-web-editor PUBLIC PkgConfig::WEBKIT PkgConfig::JSON_GLIB ${CMAKE_DL_LIBS})
else()
    message(FATAL_ERROR "No native editor backend for this platform")
endif()

        add_test(NAME ravet-web-paint COMMAND ravet-web-paint "${CMAKE_CURRENT_BINARY_DIR}/ravet-web-paint.png")
    endif()

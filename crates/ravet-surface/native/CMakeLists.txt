cmake_minimum_required(VERSION 3.22)
project(RavetWebEditor LANGUAGES CXX)
option(RAVET_WEB_EDITOR_TESTS "Build the native main-loop lifecycle probe" OFF)
if(APPLE)
    enable_language(OBJCXX)
    add_library(ravet-web-editor STATIC RavetWebView.mm RavetFileDialog.mm)
    target_compile_options(ravet-web-editor PRIVATE -fobjc-arc)
    target_link_libraries(ravet-web-editor PUBLIC "-framework Cocoa" "-framework WebKit" "-framework UniformTypeIdentifiers")
elseif(WIN32)
    set(RAVET_WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet SDK")
    if(NOT EXISTS "${RAVET_WEBVIEW2_SDK}/build/native/include/WebView2.h")
        message(FATAL_ERROR "Set RAVET_WEBVIEW2_SDK to the extracted WebView2 SDK")
    endif()
    add_library(ravet-web-editor STATIC RavetWebViewWindows.cpp RavetFileDialogWindows.cpp)
    target_include_directories(ravet-web-editor PRIVATE "${RAVET_WEBVIEW2_SDK}/build/native/include")
    if(MINGW)
        target_include_directories(ravet-web-editor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mingw")
    endif()
    target_compile_definitions(ravet-web-editor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(ravet-web-editor PUBLIC ole32 user32 shell32 shlwapi uuid comdlg32)
elseif(UNIX)
    option(RAVET_WEB_LINUX_SOFTWARE_RENDERING "Use non-accelerated WebKit painting for Linux editors" OFF)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1>=2.40)
    pkg_check_modules(JSON_GLIB REQUIRED IMPORTED_TARGET json-glib-1.0)
    add_library(ravet-web-editor STATIC RavetWebViewLinux.cpp RavetFileDialogLinux.cpp)
    target_compile_definitions(ravet-web-editor PRIVATE RAVET_WEB_LINUX_SOFTWARE_RENDERING=$<BOOL:${RAVET_WEB_LINUX_SOFTWARE_RENDERING}>)
    target_link_libraries(ravet-web-editor PUBLIC PkgConfig::WEBKIT PkgConfig::JSON_GLIB ${CMAKE_DL_LIBS})
else()
    message(FATAL_ERROR "No native editor backend for this platform")
endif()
if(RAVET_WEB_EDITOR_TESTS AND NOT APPLE)
    add_executable(ravet-web-lifecycle tests/portable.cpp)
    target_link_libraries(ravet-web-lifecycle PRIVATE ravet-web-editor)
    if(WIN32)
        add_custom_command(TARGET ravet-web-lifecycle POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RAVET_WEBVIEW2_SDK}/build/native/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:ravet-web-lifecycle>/WebView2Loader.dll")
    endif()
    enable_testing()
    add_test(NAME ravet-web-lifecycle COMMAND ravet-web-lifecycle "${CMAKE_CURRENT_SOURCE_DIR}/tests/portable.html")
    if(UNIX)
        add_library(ravet-host-gtk-shim MODULE tests/linux-host-gtk-shim.cpp)
        set_target_properties(ravet-host-gtk-shim PROPERTIES OUTPUT_NAME Swell)
        target_link_libraries(ravet-host-gtk-shim PRIVATE PkgConfig::WEBKIT)
        add_executable(ravet-host-gtk tests/linux-host-gtk.cpp)
        target_link_libraries(ravet-host-gtk PRIVATE ravet-web-editor)
        add_test(NAME ravet-host-gtk-supported COMMAND ravet-host-gtk "$<TARGET_FILE:ravet-host-gtk-shim>" supported)
        add_test(NAME ravet-host-gtk-unsupported COMMAND ravet-host-gtk "$<TARGET_FILE:ravet-host-gtk-shim>" unsupported)
        add_executable(ravet-web-paint tests/linux-paint.cpp)
        target_link_libraries(ravet-web-paint PRIVATE ravet-web-editor)
        add_test(NAME ravet-web-paint COMMAND ravet-web-paint "${CMAKE_CURRENT_BINARY_DIR}/ravet-web-paint.png")
    endif()
endif()
set_target_properties(ravet-web-editor PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ravet-web-editor PUBLIC cxx_std_17)
target_include_directories(ravet-web-editor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(RAVET_WEB_EDITOR_TESTS AND APPLE)
    add_executable(ravet-web-lifecycle tests/lifecycle.mm)
    target_compile_options(ravet-web-lifecycle PRIVATE -fobjc-arc)
    target_link_libraries(ravet-web-lifecycle PRIVATE ravet-web-editor)
    enable_testing()
    add_test(NAME ravet-web-lifecycle COMMAND ravet-web-lifecycle "${CMAKE_CURRENT_SOURCE_DIR}/tests/editor.html")
    add_executable(ravet-web-first-mouse tests/first-mouse.mm)
    target_compile_options(ravet-web-first-mouse PRIVATE -fobjc-arc)
    target_link_libraries(ravet-web-first-mouse PRIVATE ravet-web-editor)
    add_test(NAME ravet-web-first-mouse COMMAND ravet-web-first-mouse)
    add_executable(ravet-file-dialog tests/file-dialog.mm)
    target_compile_options(ravet-file-dialog PRIVATE -fobjc-arc)
    target_link_libraries(ravet-file-dialog PRIVATE ravet-web-editor)
    add_test(NAME ravet-file-dialog COMMAND ravet-file-dialog)
endif()

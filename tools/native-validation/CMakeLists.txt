cmake_minimum_required(VERSION 3.25)
project(ravet_native_contracts LANGUAGES C CXX)
if(NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Set RAVET_GAIN_LIBRARY to the prepared Rust static library")
endif()
add_executable(ravet_gain64_contracts gain64.cpp)
target_compile_features(ravet_gain64_contracts PRIVATE cxx_std_17)
target_include_directories(ravet_gain64_contracts PRIVATE ../../examples/gain-native/include)
target_link_libraries(ravet_gain64_contracts PRIVATE "${RAVET_GAIN_LIBRARY}")
if(APPLE)
    target_link_libraries(ravet_gain64_contracts PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
endif()
enable_testing()
add_test(NAME ravet_gain64_contracts COMMAND ravet_gain64_contracts)

option(RAVET_VALIDATE_WEB_EDITOR "Validate macOS Rust editor ABI (static library requires native-editor feature)" OFF)
if(APPLE AND RAVET_VALIDATE_WEB_EDITOR)
    enable_language(OBJCXX)
    add_executable(ravet_editor_abi_contracts editor.cpp ../editor-smoke/src/probe.mm)
    target_compile_features(ravet_editor_abi_contracts PRIVATE cxx_std_17)
    set_source_files_properties(../editor-smoke/src/probe.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_include_directories(ravet_editor_abi_contracts PRIVATE ../../crates/ravet-webview/include)
    target_link_libraries(ravet_editor_abi_contracts PRIVATE "${RAVET_GAIN_LIBRARY}" "-framework Cocoa" "-framework WebKit" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
    target_link_libraries(ravet_gain64_contracts PRIVATE "-framework Cocoa" "-framework WebKit")
    target_link_libraries(ravet_gain_program_contracts PRIVATE "-framework Cocoa" "-framework WebKit")
    target_link_libraries(ravet_gain_meter_contracts PRIVATE "-framework Cocoa" "-framework WebKit")
    add_test(NAME ravet_editor_abi_contracts COMMAND ravet_editor_abi_contracts)
endif()

# The reusable native headers are consumable independently of product symbols.
foreach(record_language IN ITEMS c cpp)
    add_executable(ravet_native_records_${record_language} native_records.${record_language})
    target_include_directories(ravet_native_records_${record_language} PRIVATE ../../crates/ravet-native/include ../../examples/gain-native/include)
    set_target_properties(ravet_native_records_${record_language} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
    add_test(NAME ravet_native_records_${record_language} COMMAND ravet_native_records_${record_language})
endforeach()

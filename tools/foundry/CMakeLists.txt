cmake_minimum_required(VERSION 3.22)
project(RavetFoundry LANGUAGES CXX)
get_filename_component(RAVET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
if(APPLE)
    # macOS uses ravet-webview::StandaloneWindow directly from Rust.
    add_custom_target(ravet-foundry ALL
        COMMAND "${CARGO_EXECUTABLE}" build --release --locked -p ravet-foundry --jobs 2
        WORKING_DIRECTORY "${RAVET_ROOT}" VERBATIM)
    return()
endif()
if(WIN32 AND (NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8))
    message(FATAL_ERROR "Foundry's Windows launcher currently requires native x64 MSVC and Rust x86_64-pc-windows-msvc")
endif()
set(FOUNDRY_RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/rust-target")
if(WIN32)
    set(FOUNDRY_LIBRARY "${FOUNDRY_RUST_TARGET}/release/ravet_foundry.lib")
else()
    set(FOUNDRY_LIBRARY "${FOUNDRY_RUST_TARGET}/release/libravet_foundry.a")
endif()
# Cargo performs its own precise incremental dependency checking on every build.
add_custom_target(foundry-rust ALL
    COMMAND "${CARGO_EXECUTABLE}" build --release --locked -p ravet-foundry --jobs 2 --target-dir "${FOUNDRY_RUST_TARGET}"
    WORKING_DIRECTORY "${RAVET_ROOT}" BYPRODUCTS "${FOUNDRY_LIBRARY}" VERBATIM)
add_library(foundry-engine STATIC IMPORTED)
set_target_properties(foundry-engine PROPERTIES IMPORTED_LOCATION "${FOUNDRY_LIBRARY}")
add_dependencies(foundry-engine foundry-rust)
add_subdirectory("${RAVET_ROOT}/crates/ravet-surface/native" "${CMAKE_CURRENT_BINARY_DIR}/ravet-native")
add_executable(ravet-foundry-gui native/main.cpp native/DirectoryPicker.cpp)
target_compile_features(ravet-foundry-gui PRIVATE cxx_std_17)
target_link_libraries(ravet-foundry-gui PRIVATE foundry-engine ravet-web-editor)
if(WIN32)
    target_compile_definitions(ravet-foundry-gui PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(ravet-foundry-gui PRIVATE ws2_32 bcrypt ntdll userenv)
    add_custom_command(TARGET ravet-foundry-gui POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RAVET_WEBVIEW2_SDK}/build/native/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:ravet-foundry-gui>/WebView2Loader.dll")
else()
    find_package(Threads REQUIRED)
    target_link_libraries(ravet-foundry-gui PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
install(TARGETS ravet-foundry-gui RUNTIME DESTINATION bin)
if(WIN32)
    install(FILES "${RAVET_WEBVIEW2_SDK}/build/native/x64/WebView2Loader.dll" DESTINATION bin)
endif()

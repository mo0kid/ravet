cmake_minimum_required(VERSION 3.25)
project(ravet_mobile_contracts LANGUAGES CXX)
if(NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Set RAVET_GAIN_LIBRARY to a host Rust static library")
endif()
add_executable(ravet_mobile_contracts contracts.cpp)
target_compile_features(ravet_mobile_contracts PRIVATE cxx_std_17)
target_compile_options(ravet_mobile_contracts PRIVATE -Wall -Wextra -Werror)
target_include_directories(ravet_mobile_contracts PRIVATE include ../../native/android ../../native/common ../../crates/ravet-native/include)
target_link_libraries(ravet_mobile_contracts PRIVATE "${RAVET_GAIN_LIBRARY}")
find_package(Threads REQUIRED)
target_link_libraries(ravet_mobile_contracts PRIVATE Threads::Threads)
if(APPLE)
    target_link_libraries(ravet_mobile_contracts PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
endif()
enable_testing()
add_test(NAME ravet_mobile_contracts COMMAND ravet_mobile_contracts)

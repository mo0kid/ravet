cmake_minimum_required(VERSION 3.25)
project(ravet_performance_aax VERSION 0.0.1 LANGUAGES C CXX)
set(RAVET_AAX_INTERNAL_PRECISION "32" CACHE STRING "Native processing precision behind host float32 AAX audio: 32 or 64")
set_property(CACHE RAVET_AAX_INTERNAL_PRECISION PROPERTY STRINGS 32 64)
if(NOT RAVET_AAX_INTERNAL_PRECISION MATCHES "^(32|64)$")
    message(FATAL_ERROR "RAVET_AAX_INTERNAL_PRECISION must be 32 or 64")
endif()
if(NOT APPLE)
    message(FATAL_ERROR "The local Performance AAX reference currently requires macOS")
endif()
if(NOT RAVET_VST3_SDK OR NOT RAVET_AAX_SDK OR NOT EXISTS "${RAVET_PERFORMANCE_LIBRARY}")
    message(FATAL_ERROR "External VST3/AAX SDKs and public Performance native archive are required")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_AUV2_BUILDS OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_AAX_BUILDS OFF CACHE BOOL "" FORCE)
set(SMTG_AAX_SDK_PATH "${RAVET_AAX_SDK}" CACHE PATH "" FORCE)
add_subdirectory("${RAVET_VST3_SDK}" sdk-build)
set(public_sdk_SOURCE_DIR "${RAVET_VST3_SDK}/public.sdk")
list(APPEND CMAKE_MODULE_PATH "${RAVET_VST3_SDK}/cmake/modules")
include(SMTG_AddAAXLibrary)
smtg_add_aaxplugin(ravet_performance_aax ../performance-vst3/src/performance.cpp description.cpp native-registration.cpp)
smtg_target_add_library_main(ravet_performance_aax)
target_sources(ravet_performance_aax PRIVATE "${RAVET_VST3_SDK}/public.sdk/source/vst/vstaudioeffect.cpp" "${RAVET_VST3_SDK}/public.sdk/source/vst/vsteditcontroller.cpp")
target_compile_features(ravet_performance_aax PRIVATE cxx_std_17)
target_compile_definitions(ravet_performance_aax PRIVATE RAVET_AAX_NATIVE_SAMPLE_BITS=${RAVET_AAX_INTERNAL_PRECISION})
target_include_directories(ravet_performance_aax PRIVATE "${RAVET_VST3_SDK}" "${RAVET_AAX_SDK}/Interfaces" "${RAVET_AAX_SDK}/Interfaces/ACF" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../performance-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/ravet-native/include")
target_link_libraries(ravet_performance_aax PRIVATE sdk aax_wrapper "${RAVET_PERFORMANCE_LIBRARY}" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
smtg_target_set_bundle(ravet_performance_aax BUNDLE_IDENTIFIER "org.ravet.performance.aax" COMPANY_NAME "RAVET")
enable_testing()
foreach(target IN ITEMS ravet_aax_note_input ravet_aax_note_input_fast ravet_aax_instrument_model ravet_aax_instrument_model_fast ravet_aax_instrument_registered_contracts)
    if(target STREQUAL "ravet_aax_instrument_registered_contracts")
        set(source registered-instrument.cpp)
    elseif(target MATCHES "instrument_model")
        set(source instrument-model.cpp)
    else()
        set(source note-input.cpp)
    endif()
    add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/${source}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/description-stub.cpp")
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE RAVET_AAX_NATIVE_SAMPLE_BITS=${RAVET_AAX_INTERNAL_PRECISION})
    target_include_directories(${target} PRIVATE "${RAVET_AAX_SDK}/Interfaces" "${RAVET_AAX_SDK}/Interfaces/ACF" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../performance-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/ravet-native/include")
    target_link_libraries(${target} PRIVATE aax_wrapper sdk "${RAVET_PERFORMANCE_LIBRARY}" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
    if(target MATCHES "_fast$" AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -ffast-math)
    endif()
    if(NOT target STREQUAL "ravet_aax_instrument_registered_contracts")
        add_test(NAME ${target} COMMAND ${target})
    endif()
endforeach()

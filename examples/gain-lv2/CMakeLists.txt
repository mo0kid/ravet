cmake_minimum_required(VERSION 3.25)
project(ravet_gain_lv2 LANGUAGES CXX)
if(NOT EXISTS "${RAVET_LV2_SDK}/include/lv2/core/lv2.h" OR NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Supply external RAVET_LV2_SDK and Rust RAVET_GAIN_LIBRARY")
endif()
add_library(ravet_gain_lv2 MODULE gain.cpp)
target_compile_features(ravet_gain_lv2 PRIVATE cxx_std_17)
target_include_directories(ravet_gain_lv2 PRIVATE "${RAVET_LV2_SDK}/include" ../gain-native/include)
target_link_libraries(ravet_gain_lv2 PRIVATE "${RAVET_GAIN_LIBRARY}")
set_target_properties(ravet_gain_lv2 PROPERTIES PREFIX "" SUFFIX ".so")
if(APPLE)
    target_link_libraries(ravet_gain_lv2 PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
elseif(UNIX)
    target_link_libraries(ravet_gain_lv2 PRIVATE pthread dl m)
endif()
add_executable(ravet_lv2_contracts ../../tools/lv2-validation/contracts.cpp)
target_compile_features(ravet_lv2_contracts PRIVATE cxx_std_17)
target_include_directories(ravet_lv2_contracts PRIVATE "${RAVET_LV2_SDK}/include")
target_link_libraries(ravet_lv2_contracts PRIVATE ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME ravet_lv2_contracts COMMAND ravet_lv2_contracts $<TARGET_FILE:ravet_gain_lv2>)

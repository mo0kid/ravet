cmake_minimum_required(VERSION 3.25)
project(ravet_gain_lv2 LANGUAGES CXX)
if(NOT EXISTS "${RAVET_LV2_SDK}/include/lv2/core/lv2.h" OR NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Supply external RAVET_LV2_SDK and Rust RAVET_GAIN_LIBRARY")
endif()
add_library(ravet_gain_lv2 MODULE gain.cpp)
target_compile_features(ravet_gain_lv2 PRIVATE cxx_std_17)
target_include_directories(ravet_gain_lv2 PRIVATE "${RAVET_LV2_SDK}/include" ../../crates/ravet-native/include)
target_link_libraries(ravet_gain_lv2 PRIVATE "${RAVET_GAIN_LIBRARY}")
set_target_properties(ravet_gain_lv2 PROPERTIES PREFIX "" SUFFIX ".so")
if(APPLE)
    target_link_libraries(ravet_gain_lv2 PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
elseif(UNIX)
    target_link_libraries(ravet_gain_lv2 PRIVATE pthread dl m)
endif()
add_executable(ravet_lv2_contracts ../../tools/lv2-validation/contracts.cpp)
target_compile_features(ravet_lv2_contracts PRIVATE cxx_std_17)
target_include_directories(ravet_lv2_contracts PRIVATE "${RAVET_LV2_SDK}/include")
target_link_libraries(ravet_lv2_contracts PRIVATE ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(ravet_lv2_contracts PRIVATE pthread)
endif()
enable_testing()
add_test(NAME ravet_lv2_contracts COMMAND ravet_lv2_contracts $<TARGET_FILE:ravet_gain_lv2>)
add_executable(ravet_lv2_state_contracts ../../tools/lv2-validation/state.cpp)
target_compile_features(ravet_lv2_state_contracts PRIVATE cxx_std_17)
target_include_directories(ravet_lv2_state_contracts PRIVATE "${RAVET_LV2_SDK}/include" ../../crates/ravet-native/include)
target_link_libraries(ravet_lv2_state_contracts PRIVATE ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(ravet_lv2_state_contracts PRIVATE pthread)
endif()
add_test(NAME ravet_lv2_state_contracts COMMAND ravet_lv2_state_contracts $<TARGET_FILE:ravet_gain_lv2>)

if(RAVET_ROUTING_LIBRARY)
    if(NOT EXISTS "${RAVET_ROUTING_LIBRARY}")
        message(FATAL_ERROR "Supply an existing Rust RAVET_ROUTING_LIBRARY")
    endif()
    add_library(ravet_note_routing_lv2 MODULE ../note-routing-lv2/routing.cpp)
    target_compile_features(ravet_note_routing_lv2 PRIVATE cxx_std_17)
    target_include_directories(ravet_note_routing_lv2 PRIVATE "${RAVET_LV2_SDK}/include" ../../crates/ravet-native/include ../bus-routing-native/include)
    target_link_libraries(ravet_note_routing_lv2 PRIVATE "${RAVET_ROUTING_LIBRARY}")
    set_target_properties(ravet_note_routing_lv2 PROPERTIES PREFIX "" SUFFIX ".so")
    foreach(target ravet_lv2_output_contracts ravet_lv2_output_fast_contracts ravet_lv2_processor_output_contracts ravet_lv2_processor_output_fast_contracts ravet_lv2_routing_contracts)
        if(target STREQUAL "ravet_lv2_routing_contracts")
            set(fixture ../../tools/lv2-validation/routing.cpp)
        elseif(target MATCHES "processor_output")
            set(fixture ../../tools/lv2-validation/processor-output.cpp)
        else()
            set(fixture ../../tools/lv2-validation/output.cpp)
        endif()
        add_executable(${target} ${fixture})
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_include_directories(${target} PRIVATE "${RAVET_LV2_SDK}/include" ../../crates/ravet-native/include ../bus-routing-native/include)
        target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
        if(target MATCHES "processor_output")
            target_link_libraries(${target} PRIVATE "${RAVET_ROUTING_LIBRARY}")
        endif()
        if(target MATCHES "fast_contracts")
            target_compile_options(${target} PRIVATE -O3 -ffast-math)
        endif()
        if(target STREQUAL "ravet_lv2_routing_contracts")
            add_test(NAME ${target} COMMAND ${target} $<TARGET_FILE:ravet_note_routing_lv2>)
        else()
            add_test(NAME ${target} COMMAND ${target})
        endif()
    endforeach()
    if(APPLE)
        foreach(target ravet_note_routing_lv2 ravet_lv2_processor_output_contracts ravet_lv2_processor_output_fast_contracts)
            target_link_libraries(${target} PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" iconv)
        endforeach()
    elseif(UNIX)
        foreach(target ravet_note_routing_lv2 ravet_lv2_processor_output_contracts ravet_lv2_processor_output_fast_contracts)
            target_link_libraries(${target} PRIVATE pthread dl m)
        endforeach()
    endif()
endif()

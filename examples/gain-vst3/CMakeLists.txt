        message(FATAL_ERROR "Stock AAX profile only supports internal precision 32")
    endif()

    smtg_add_aaxplugin(ravet_gain_aax src/gain.cpp ../gain-aax/description.cpp)

        set(RAVET_AAX_REFERENCE_PROFILE native CACHE STRING "Local AAX reference implementation: native or stock")
    endif()
    if(RAVET_AAX_REFERENCE_PROFILE STREQUAL "native")
        target_sources(ravet_gain_aax PRIVATE ../gain-aax/native-registration.cpp)
        target_compile_definitions(ravet_gain_aax PRIVATE RAVET_AAX_NATIVE_SAMPLE_BITS=${RAVET_AAX_INTERNAL_PRECISION})
        include("${CMAKE_CURRENT_SOURCE_DIR}/../gain-aax/native-owner.cmake")
        if(RAVET_AAX_INTERNAL_PRECISION STREQUAL "64")
            include("${CMAKE_CURRENT_SOURCE_DIR}/../gain-aax/native-precision.cmake")
        endif()
        if(RAVET_AAX_EDITOR)
            include("${CMAKE_CURRENT_SOURCE_DIR}/../gain-aax/native-editor.cmake")
        endif()
    elseif(RAVET_AAX_REFERENCE_PROFILE STREQUAL "stock")
        if(RAVET_AAX_EDITOR)
            message(FATAL_ERROR "RAVET_AAX_EDITOR requires the native Gain profile")
        endif()
        include("${CMAKE_CURRENT_SOURCE_DIR}/../gain-aax/native-parameters.cmake")
    else()
        message(FATAL_ERROR "RAVET_AAX_REFERENCE_PROFILE must be native or stock")
    endif()
    smtg_target_add_library_main(ravet_gain_aax)
    target_sources(ravet_gain_aax PRIVATE "${RAVET_VST3_SDK}/public.sdk/source/vst/vstaudioeffect.cpp" "${RAVET_VST3_SDK}/public.sdk/source/vst/vsteditcontroller.cpp")

    target_link_libraries(ravet_gain_aax PRIVATE sdk aax_wrapper "${RAVET_GAIN_LIBRARY}")
    if(APPLE)
        target_link_libraries(ravet_gain_aax PRIVATE "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
        smtg_target_set_bundle(ravet_gain_aax BUNDLE_IDENTIFIER "org.ravet.gain.aax" COMPANY_NAME "RAVET")
    endif()

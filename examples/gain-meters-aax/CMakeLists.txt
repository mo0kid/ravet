cmake_minimum_required(VERSION 3.25)
project(ravet_gain_meters_aax VERSION 0.0.1 LANGUAGES C CXX)
set(RAVET_AAX_INTERNAL_PRECISION "32" CACHE STRING "Native processing precision behind host float32 AAX audio: 32 or 64")
set_property(CACHE RAVET_AAX_INTERNAL_PRECISION PROPERTY STRINGS 32 64)
if(NOT RAVET_AAX_INTERNAL_PRECISION MATCHES "^(32|64)$")
    message(FATAL_ERROR "RAVET_AAX_INTERNAL_PRECISION must be 32 or 64")
endif()
if(NOT APPLE)
    message(FATAL_ERROR "The local Gain meter AAX reference currently requires macOS")
endif()
if(NOT RAVET_VST3_SDK OR NOT RAVET_AAX_SDK OR NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "External VST3/AAX SDKs and public Gain native archive are required")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_AUV2_BUILDS OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_AAX_BUILDS OFF CACHE BOOL "" FORCE)
set(SMTG_AAX_SDK_PATH "${RAVET_AAX_SDK}" CACHE PATH "" FORCE)
add_subdirectory("${RAVET_VST3_SDK}" sdk-build)
set(public_sdk_SOURCE_DIR "${RAVET_VST3_SDK}/public.sdk")
list(APPEND CMAKE_MODULE_PATH "${RAVET_VST3_SDK}/cmake/modules")
include(SMTG_AddAAXLibrary)
smtg_add_aaxplugin(ravet_gain_meters_aax bootstrap.cpp description.cpp native-registration.cpp)
smtg_target_add_library_main(ravet_gain_meters_aax)
target_compile_features(ravet_gain_meters_aax PRIVATE cxx_std_17)
target_compile_definitions(ravet_gain_meters_aax PRIVATE RAVET_AAX_NATIVE_SAMPLE_BITS=${RAVET_AAX_INTERNAL_PRECISION})
target_include_directories(ravet_gain_meters_aax PRIVATE "${RAVET_VST3_SDK}" "${RAVET_AAX_SDK}/Interfaces" "${RAVET_AAX_SDK}/Interfaces/ACF" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../gain-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/ravet-native/include")
target_link_libraries(ravet_gain_meters_aax PRIVATE sdk aax_wrapper "${RAVET_GAIN_LIBRARY}" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
smtg_target_set_bundle(ravet_gain_meters_aax BUNDLE_IDENTIFIER "org.ravet.gain_meters.aax" COMPANY_NAME "RAVET")
enable_testing()
foreach(target IN ITEMS ravet_aax_meters ravet_aax_meters_fast ravet_aax_meters_registered_contracts)
 if(target STREQUAL "ravet_aax_meters_registered_contracts")
  set(source registered-meters.cpp)
 else()
  set(source meters.cpp)
 endif()
 add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/${source}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/description-stub.cpp")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_definitions(${target} PRIVATE RAVET_AAX_NATIVE_SAMPLE_BITS=${RAVET_AAX_INTERNAL_PRECISION})
 target_include_directories(${target} PRIVATE "${RAVET_VST3_SDK}" "${RAVET_AAX_SDK}/Interfaces" "${RAVET_AAX_SDK}/Interfaces/ACF" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../gain-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/ravet-native/include")
 target_link_libraries(${target} PRIVATE aax_wrapper sdk "${RAVET_GAIN_LIBRARY}" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
 if(target MATCHES "_fast$")
  target_compile_options(${target} PRIVATE -ffast-math)
 endif()
 if(NOT target STREQUAL "ravet_aax_meters_registered_contracts")
  add_test(NAME ${target} COMMAND ${target})
 endif()
endforeach()

if(RAVET_METER_PERFORMANCE_LIBRARY AND RAVET_METER_ROUTING_LIBRARY)
 foreach(target IN ITEMS ravet_aax_meter_composition ravet_aax_meter_composition_fast)
  add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/meter-composition.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/aax-validation/description-stub.cpp")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${RAVET_VST3_SDK}" "${RAVET_AAX_SDK}/Interfaces" "${RAVET_AAX_SDK}/Interfaces/ACF" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../gain-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../performance-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../bus-routing-native/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/ravet-native/include")
  target_link_libraries(${target} PRIVATE aax_wrapper sdk "${RAVET_GAIN_LIBRARY}" "${RAVET_METER_PERFORMANCE_LIBRARY}" "${RAVET_METER_ROUTING_LIBRARY}" "-framework Security" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Carbon" "-framework Cocoa" iconv)
  if(target MATCHES "_fast$")
   target_compile_options(${target} PRIVATE -ffast-math)
  endif()
  add_test(NAME ${target} COMMAND ${target})
 endforeach()
endif()

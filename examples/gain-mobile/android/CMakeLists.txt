cmake_minimum_required(VERSION 3.25)
project(ravet_android LANGUAGES CXX)
if(NOT ANDROID OR NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Requires Android NDK toolchain and target RAVET_GAIN_LIBRARY")
endif()
add_library(ravet_gain_mobile SHARED native.cpp)
target_compile_features(ravet_gain_mobile PRIVATE cxx_std_17)
target_compile_options(ravet_gain_mobile PRIVATE -Wall -Wextra -Werror)
target_include_directories(ravet_gain_mobile PRIVATE ../../../native/common ../../../native/android ../../../crates/ravet-native/include)
target_link_libraries(ravet_gain_mobile PRIVATE "${RAVET_GAIN_LIBRARY}" aaudio log android)
target_link_options(ravet_gain_mobile PRIVATE "-Wl,-z,max-page-size=16384")

cmake_minimum_required(VERSION 3.25)
project(ravet_auv3 LANGUAGES CXX OBJCXX)
if(NOT APPLE OR NOT EXISTS "${RAVET_GAIN_LIBRARY}")
    message(FATAL_ERROR "Requires Apple SDK and RAVET_GAIN_LIBRARY for this target")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
add_library(ravet_auv3 STATIC ../../native/apple/RavetAudioUnit.mm ../../native/apple/RavetParameterViewController.mm GainAudioUnit.mm)
target_include_directories(ravet_auv3 PUBLIC . ../../native/apple ../../native/common ../../crates/ravet-native/include)
target_compile_options(ravet_auv3 PRIVATE -fobjc-arc -fapplication-extension -Wall -Wextra -Werror)
target_link_libraries(ravet_auv3 PUBLIC "${RAVET_GAIN_LIBRARY}" "-framework AudioToolbox" "-framework AVFoundation" "-framework CoreMIDI" "-framework CoreAudioKit" "-framework Foundation" "-framework Security" iconv)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(ravet_auv3 PUBLIC "-framework UIKit")
else()
    add_executable(RavetGainContainer MACOSX_BUNDLE Container.cpp)
    set_target_properties(RavetGainContainer PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Container.plist.in")
    add_dependencies(RavetGainContainer RavetGainExtension)
    add_custom_target(embed_ravet_auv3 ALL DEPENDS RavetGainContainer RavetGainExtension
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:RavetGainContainer>/Contents/PlugIns"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:RavetGainExtension>" "$<TARGET_BUNDLE_DIR:RavetGainContainer>/Contents/PlugIns/RavetGainExtension.appex")
    target_link_libraries(ravet_auv3 PUBLIC "-framework Cocoa")
endif()
add_executable(RavetGainExtension MACOSX_BUNDLE Extension.mm)
set_target_properties(RavetGainExtension PROPERTIES BUNDLE_EXTENSION appex MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
target_compile_options(RavetGainExtension PRIVATE -fobjc-arc -fapplication-extension -Wall -Wextra -Werror)
target_link_libraries(RavetGainExtension PRIVATE ravet_auv3)
target_link_options(RavetGainExtension PRIVATE "-Wl,-e,_NSExtensionMain")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    option(RAVET_EMBED_AUV3_CONTRACTS "Include explicit simulator contract runner in public fixture app" OFF)
    add_executable(RavetGainMobile MACOSX_BUNDLE ../gain-mobile/apple/main.mm ../../native/apple/RavetMobileAudio.mm)
    set_target_properties(RavetGainMobile PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/../gain-mobile/apple/Info.plist.in")
    target_compile_options(RavetGainMobile PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    target_link_libraries(RavetGainMobile PRIVATE ravet_auv3 "-framework UIKit" "-framework AVKit")
    if(RAVET_EMBED_AUV3_CONTRACTS)
        target_sources(RavetGainMobile PRIVATE ../../tools/auv3-validation/contracts.mm)
        target_compile_definitions(RavetGainMobile PRIVATE RAVET_AUV3_CONTRACTS_EMBEDDED=1)
    endif()
    add_dependencies(RavetGainMobile RavetGainExtension)
    add_custom_target(embed_ravet_auv3 ALL DEPENDS RavetGainMobile RavetGainExtension
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:RavetGainMobile>/PlugIns"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:RavetGainExtension>" "$<TARGET_BUNDLE_DIR:RavetGainMobile>/PlugIns/RavetGainExtension.appex")
else()
    add_executable(ravet_auv3_contracts ../../tools/auv3-validation/contracts.mm)
    target_compile_options(ravet_auv3_contracts PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    target_link_libraries(ravet_auv3_contracts PRIVATE ravet_auv3)
    enable_testing()
    add_test(NAME ravet_auv3_contracts COMMAND ravet_auv3_contracts)
endif()
